def identity(path):
    with path.open('rb') as stream:
        size = path.stat().st_size
        digest = 'sha256:' + hashlib.file_digest(stream, 'sha256').hexdigest()
    return {'digest': digest, 'size': size}

require(receipt['status'] == 'normal-source-native-build-passed'
        and receipt['sourceRevision'] == '907e8fa0791e85d764d7b06a716f4f238a84e05d',

evidence = root / 'node-attempt4'
evidence.mkdir(mode=0o700)
result = {'schemaVersion': 'lsf.dotnet.memorypack.node-observation.v1', 'status': 'in-progress',
    'sourceRevision': receipt['sourceRevision'],

    'compilerPayloadMounted': False, 'sdkSourceRevision': 'c7716d822132297e511ab2fc48279c93d6a94eae',
    'adapterBuildSourceRevision': '260ca8aead1a851a873162ff1138819c21c78025', 'network': 'Docker --network none', 'invocations': [], 'samples': [],

    'limitations': ['Finite pure-library fixture; no default Task/ThreadPool/HttpClient qualification.',
                    'Linux /proc samples are non-atomic; no production sizing or complete GC claim.']}
client = node = None
stage = 'node-input-separation'
try:
    require(not any(Path(name).exists() for name in ('/capture','/public','/adapters')), 'compiler-payload-must-be-unavailable-at-node-invocation')

    release_set = read_json(root / 'releases/release-set.json')

        stage = 'normal-node-start'
        node = connect(client, native / 'latentd', node_directory, config, 'examples', 1)
        result['startup'] = node.startup_record
        probe = Probe(node, file_identity(native / 'latentd'))

            '--expected-generation', '0')

                == release_set['releases'][0]['componentDigest'], 'signed-publication-component')
        result['publication'] = published
        publication = published['data']['operation']['publication']['id']

        result['samples'].append(sample(client, probe, 'signed-dormant', 1))
        stage = 'clock-grant-denial'
        denied = call(client, target, 'greeting', 'greet', ['Ada'], 'memorypack-clock-denied')

                'clock-import-requires-explicit-grant')

        installed = [row for row in node.startup_record['providers'] if row['id'] == 'clockMonotonic']
        require(len(installed) == 1 and installed[0]['capability'] == 'latent:clock/monotonic@0.1.0'
                and installed[0]['profile'] == 'activation-monotonic-v1', 'normal-clock-provider')
        descriptor = installed[0]

        client.call('policy', '--kind', 'provider-binding', 'apply', '--id', 'clockMonotonic-installed',

            'id': 'runtime', 'effect': 'allow',

            for _ in range(3):
                result['samples'].append(sample(client, probe, 'dormant-' + str(count), len(names)))
        os_rows = [row['os']['metrics'] for row in result['samples'] if row['phase'].startswith('dormant-')]
        require(all(row['processes'] == 1 and row['listeners'] == 1 and row['udpSockets'] == 0 for row in os_rows),

        node = None
        result['cliCalls'] = client.calls
        cancellation.check()
    require(all(identity(native / name) == receipt['files'][name]
                for name in ('latent', 'latentd', 'capsule_authoring')), 'normal-native-inputs-changed')
    result['status'] = 'passed'
    encoded = json.dumps(result, separators=(',', ':')).encode('utf-8')

    raise

        client.node = None
    if node is not None:

                   'scope': 'finally block after the retained original node observation'}
        try:
            with node.cancellation.defer():
                node.stop()
                cleanup['physicalShutdown'] = stopped_record(node)
            cleanup['status'] = 'normal-shutdown-passed'
        except BaseException as cleanup_error:
            cleanup['status'], cleanup['reason'] = 'cleanup-unconfirmed', str(cleanup_error)
            try:
                node.close()
            except BaseException as close_error:
                cleanup['closeFailure'] = str(close_error)
        write_json(evidence / 'PHYSICAL-CLEANUP.json', cleanup)
